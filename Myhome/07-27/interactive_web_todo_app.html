원본 파이썬 코드 및 웹 이식 구조

입력하신 파이썬 데이터 구조가 웹 자바스크립트로 어떻게 대응되는지 비교할 수 있습니다.

원본 Python 코드
todo.py 
# 파이썬 Todo 데이터 구조
todos = []

def add_todo(task):
    todos.append({'task': task, 'completed': False})

def view_todos():
    for i, item in enumerate(todos, 1):
        status = "[X]" if item['completed'] else "[ ]"
        print(f"{i}. {status} {item['task']}")

def mark_completed(task_index):
    if 1 <= task_index <= len(todos):
        todos[task_index - 1]['completed'] = True

def delete_completed_tasks():
    global todos
    todos = [task for task in todos if not task['completed']]
웹 브라우저 JavaScript 대응 코드
app.js 
// 자바스크립트 Todo 데이터 구조 (브라우저 연동)
let todos = JSON.parse(localStorage.getItem('todos')) || [];

function addTodo(task) {
    todos.push({ task: task, completed: false });
    saveAndRender();
}

function toggleCompleted(index) {
    todos[index].completed = !todos[index].completed;
    saveAndRender();
}

function deleteCompletedTasks() {
    todos = todos.filter(item => !item.completed);
    saveAndRender();
}
